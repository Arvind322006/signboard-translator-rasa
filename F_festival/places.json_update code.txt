from rasa_sdk import Action, Tracker
from rasa_sdk.executor import CollectingDispatcher
import base64
import cv2
import numpy as np
import easyocr
from googletrans import Translator
import re
import json


# 📌 Load place info from JSON
with open("data/places.json", "r", encoding="utf-8") as f:
    places_data = json.load(f)


# 📌 1. Action for signboard translation + auto place info
class ActionTranslateSign(Action):
    def name(self):
        return "action_translate_sign"

    def run(self, dispatcher: CollectingDispatcher,
            tracker: Tracker,
            domain: dict):

        detected_text = ""

        # 1️⃣ Check for uploaded image (metadata)
        metadata = tracker.latest_message.get("metadata", {})
        image_data = metadata.get("image")
        if image_data:
            try:
                header, encoded = image_data.split(",", 1)
                img_bytes = base64.b64decode(encoded)
                nparr = np.frombuffer(img_bytes, np.uint8)
                img = cv2.imdecode(nparr, cv2.IMREAD_COLOR)

                reader = easyocr.Reader(['en', 'ta'])
                results = reader.readtext(img)

                if results:
                    detected_text = " ".join([res[1] for res in results])
                else:
                    dispatcher.utter_message(text="❌ Couldn’t detect any text in the image.")
                    return []

            except Exception as e:
                dispatcher.utter_message(text=f"⚠️ Error processing image: {str(e)}")
                return []

        else:
            # 2️⃣ If no image, fallback to user message
            user_message = tracker.latest_message.get("text", "").strip()
            if user_message:
                detected_text = re.sub(r"(translate|to english|in english)", "", user_message, flags=re.I).strip()

        # 3️⃣ Translate text
        if not detected_text:
            dispatcher.utter_message(text="⚠️ No text provided to translate.")
            return []

        translator = Translator()
        translated = translator.translate(detected_text, src="auto", dest="en")

        # 4️⃣ Base reply
        dispatcher.utter_message(text=f"📝 Detected: {detected_text}")
        dispatcher.utter_message(text=f"🌍 English: {translated.text}")

        # 5️⃣ Check for place info
        for place in places_data:
            if place.lower() in detected_text.lower() or place in detected_text:
                info = places_data[place]
                response = (
                    f"✨ Extra info about {place}:\n"
                    f"- Famous for: {', '.join(info.get('famous', []))}\n"
                    f"- Special: {info.get('special', 'N/A')}\n"
                    f"- Tourist places: {', '.join(info.get('tourist_places', []))}\n"
                    f"- Temples/Culture: {info.get('culture', 'N/A')}\n"
                )
                dispatcher.utter_message(text=response)
                break

        return []


# 📌 2. Action for explicit place queries
class ActionGetPlaceInfo(Action):
    def name(self):
        return "action_get_place_info"

    def run(self, dispatcher: CollectingDispatcher,
            tracker: Tracker,
            domain: dict):

        user_msg = tracker.latest_message.get("text", "").lower()
        found_place = None
        for place in places_data:
            if place.lower() in user_msg:
                found_place = place
                break

        if not found_place:
            dispatcher.utter_message(text="❌ Sorry, I don’t have information about that place yet.")
            return []

        info = places_data[found_place]
        response = None

        if "famous" in user_msg:
            response = f"{found_place} is famous for {', '.join(info['famous'])}."
        elif "special" in user_msg:
            response = f"{found_place} is special because {info['special']}."
        elif "tourist" in user_msg or "visit" in user_msg:
            response = f"Tourist attractions in {found_place} include {', '.join(info['tourist_places'])}."
        elif "food" in user_msg or "eat" in user_msg:
            response = f"Popular foods in {found_place} are {', '.join(info['food'])}."
        elif "highway" in user_msg or "road" in user_msg:
            for city, highway in info['highways'].items():
                if city.lower() in user_msg:
                    response = f"You can reach {city} from {found_place} via {highway}."
                    break
            if not response:
                response = f"Highways from {found_place} include: {info['highways']}."
        elif "climate" in user_msg or "weather" in user_msg:
            response = f"The climate in {found_place} is {info['climate']}."
        elif "transport" in user_msg or "travel" in user_msg:
            response = (
                f"In {found_place}, transport options include "
                f"railway ({info['transport']['railway']}), "
                f"airport ({info['transport']['airport']})."
            )
        elif "culture" in user_msg or "festival" in user_msg:
            response = f"The cultural highlights of {found_place} include {info['culture']}."
        else:
            response = f"Here’s some info about {found_place}: {info['overview']}"

        dispatcher.utter_message(text=response)
        return []
