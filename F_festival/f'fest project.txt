import easyocr
from googletrans import Translator
from rasa_sdk import Action, Tracker
from rasa_sdk.executor import CollectingDispatcher

class ActionTranslateSign(Action):
    def name(self):
        return "action_translate_sign"

    def run(self, dispatcher: CollectingDispatcher,
            tracker: Tracker,
            domain: dict):

        # Static image for OCR demo
        image_path = "signboard.jpg"

        # Get user’s latest text
        user_message = tracker.latest_message.get("text", "").lower().strip()

        detected_text = ""

        if any(keyword in user_message for keyword in ["translate", "to english"]):
            # User provided some text to translate
            # Remove trigger words like "translate", "to english"
            detected_text = (
                user_message.replace("translate", "")
                            .replace("to english", "")
                            .strip()
            )
        else:
            # Run OCR if no direct text is provided
            reader = easyocr.Reader(['en', 'hi', 'ta'])  # English, Hindi, Tamil
            results = reader.readtext(image_path)
            detected_text = " ".join([res[1] for res in results])

        if not detected_text:
            dispatcher.utter_message(text="❌ I couldn't detect any text. Try again.")
            return []

        # Translate detected text -> English
        translator = Translator()
        translated = translator.translate(detected_text, src="auto", dest="en")

        dispatcher.utter_message(text=f"📝 Detected: {detected_text}")
        dispatcher.utter_message(text=f"🌐 Translation: {translated.text}")
        return []

      const input = document.getElementById("userInput");
      const message = input.value.trim();
      if (!message) return;
      input.value = "";

      const chatbox = document.getElementById("chatbox");
      chatbox.innerHTML += `<div class="message user">${message}</div>`;

        const response = await fetch("http://localhost:5005/webhooks/rest/webhook", {

          body: JSON.stringify({ sender: "user", message: message })

        data.forEach(reply => {
          if (reply.text) {
            if (reply.text.startsWith("Detected:")) {

        chatbox.innerHTML += `<div class="message bot" style="background:#991b1b;color:#fff;">⚠️ Error: Could not connect to Rasa</div>`;
        chatbox.scrollTop = chatbox.scrollHeight;
      }
    }

from rasa_sdk import Action, Tracker
from rasa_sdk.executor import CollectingDispatcher
import base64
import cv2
import numpy as np
import easyocr
from googletrans import Translator

class ActionTranslateSign(Action):
    def name(self):
        return "action_translate_sign"

    def run(self, dispatcher: CollectingDispatcher,
            tracker: Tracker,
            domain: dict):

        # 👀 Get metadata (image in base64)
        metadata = tracker.latest_message.get("metadata", {})
        image_data = metadata.get("image")

        if not image_data:
            dispatcher.utter_message(text="⚠️ No image received. Please try again.")
            return []

        try:
            # ✅ Decode base64
            header, encoded = image_data.split(",", 1)
            img_bytes = base64.b64decode(encoded)
            nparr = np.frombuffer(img_bytes, np.uint8)
            img = cv2.imdecode(nparr, cv2.IMREAD_COLOR)

            # ✅ OCR
            reader = easyocr.Reader(['en', 'ta'])
            results = reader.readtext(img)

            if not results:
                dispatcher.utter_message(text="❌ Couldn’t detect any text in the image.")
                return []

            detected_texts = [res[1] for res in results]
            full_text = " ".join(detected_texts)

            # ✅ Translate
            translator = Translator()
            translated = translator.translate(full_text, src="auto", dest="en")

            # ✅ Send back BOTH original + translated
            dispatcher.utter_message(text=f"📝 Detected: {full_text}")
            dispatcher.utter_message(text=f"🌍 English: {translated.text}")

        except Exception as e:
            dispatcher.utter_message(text=f"⚠️ Error: {str(e)}")
return []

        # 1️⃣ Check for uploaded image (metadata)
        metadata = tracker.latest_message.get("metadata", {})
        image_data = metadata.get("image")

        if image_data:
            try:
                header, encoded = image_data.split(",", 1)
                img_bytes = base64.b64decode(encoded)
                nparr = np.frombuffer(img_bytes, np.uint8)
                img = cv2.imdecode(nparr, cv2.IMREAD_COLOR)

                reader = easyocr.Reader(['en','ta'])
                results = reader.readtext(img)

                if results:
                    detected_text = " ".join([res[1] for res in results])
                else:
                    dispatcher.utter_message(text="❌ Couldn’t detect any text in the image.")
                    return []

            except Exception as e:
                dispatcher.utter_message(text=f"⚠️ Error processing image: {str(e)}")
                return []

        else:
            # 2️⃣ If no image, fallback to user message
            user_message = tracker.latest_message.get("text", "").strip()
            if user_message:
                # Remove keywords like "translate", "to english"
                detected_text = re.sub(r"(translate|to english|in english)", "", user_message, flags=re.I).strip()

        # 3️⃣ Translate text
        if not detected_text:
            dispatcher.utter_message(text="⚠️ No text provided to translate.")
            return []

        translator = Translator()
        translated = translator.translate(detected_text, src="auto", dest="en")

        # 4️⃣ Reply
        dispatcher.utter_message(text=f"📝 Detected: {detected_text}")
        dispatcher.utter_message(text=f"🌍 English: {translated.text}")
        return []
