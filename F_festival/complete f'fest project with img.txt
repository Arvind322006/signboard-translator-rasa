actions/actions.py



from rasa_sdk import Action, Tracker
from rasa_sdk.executor import CollectingDispatcher
import base64
import cv2
import numpy as np
import easyocr
from googletrans import Translator
import re


class ActionTranslateSign(Action):
    def name(self):
        return "action_translate_sign"

    def run(self, dispatcher: CollectingDispatcher,
            tracker: Tracker,
            domain: dict):

        detected_text = ""

        # 1️⃣ Check for uploaded image (metadata)
        metadata = tracker.latest_message.get("metadata", {})
        image_data = metadata.get("image")

        if image_data:
            try:
                header, encoded = image_data.split(",", 1)
                img_bytes = base64.b64decode(encoded)
                nparr = np.frombuffer(img_bytes, np.uint8)
                img = cv2.imdecode(nparr, cv2.IMREAD_COLOR)

                reader = easyocr.Reader(['en','ta'])
                results = reader.readtext(img)

                if results:
                    detected_text = " ".join([res[1] for res in results])
                else:
                    dispatcher.utter_message(text="❌ Couldn’t detect any text in the image.")
                    return []

            except Exception as e:
                dispatcher.utter_message(text=f"⚠️ Error processing image: {str(e)}")
                return []

        else:
            # 2️⃣ If no image, fallback to user message
            user_message = tracker.latest_message.get("text", "").strip()
            if user_message:
                # Remove keywords like "translate", "to english"
                detected_text = re.sub(r"(translate|to english|in english)", "", user_message, flags=re.I).strip()

        # 3️⃣ Translate text
        if not detected_text:
            dispatcher.utter_message(text="⚠️ No text provided to translate.")
            return []

        translator = Translator()
        translated = translator.translate(detected_text, src="auto", dest="en")

        # 4️⃣ Reply
        dispatcher.utter_message(text=f"📝 Detected: {detected_text}")
        dispatcher.utter_message(text=f"🌍 English: {translated.text}")
        return []

  <script>
    const chatBox = document.getElementById("chat-box");
    const input = document.getElementById("user-input");

    function appendMessage(content, sender) {
      const msg = document.createElement("div");
      msg.classList.add("message", sender);
      msg.innerHTML = content;
      chatBox.appendChild(msg);
      chatBox.scrollTop = chatBox.scrollHeight;
    }

    function showTyping() {
      const typingMsg = document.createElement("div");
      typingMsg.classList.add("message", "bot", "typing");
      typingMsg.setAttribute("id", "typing");
      typingMsg.innerText = "Bot is typing...";
      chatBox.appendChild(typingMsg);
      chatBox.scrollTop = chatBox.scrollHeight;
    }

    function hideTyping() {
      const typingMsg = document.getElementById("typing");
      if (typingMsg) typingMsg.remove();
    }

    function sendMessage() {
      const text = input.value.trim();
      if (!text) return;
      appendMessage(text, "user");
      input.value = "";
      showTyping();

      fetch("http://localhost:5005/webhooks/rest/webhook", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ sender: "user", message: text })
      })
      .then(res => res.json())
      .then(data => {
        hideTyping();
        data.forEach(d => {
          if (d.text) appendMessage(d.text, "bot");
        });
      });
    }

    // 🔹 Press Enter to send
    input.addEventListener("keypress", function(e) {
      if (e.key === "Enter") {
        e.preventDefault();
        sendMessage();
      }
    });

    function uploadImage(event) {
      const file = event.target.files[0];
      if (!file) return;

      const reader = new FileReader();
      reader.onload = () => {
        appendMessage(`<img src="${reader.result}" width="200" style="border-radius:10px;"/>`, "user");

        showTyping();

        fetch("http://localhost:5005/webhooks/rest/webhook", {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({
            sender: "user",
            message: "[image]",
            metadata: { image: reader.result }
          })
        })
        .then(res => res.json())
        .then(data => {
          hideTyping();
          data.forEach(d => {
            if (d.text) appendMessage(d.text, "bot");
          });
        });
      };
      reader.readAsDataURL(file);
    }
  </script>
